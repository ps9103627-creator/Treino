  <script>
    // ESTADO DA APLICAÇÃO
    let alarmes = [];
    let metaAtual = "";
    let reconhecendo = false;

    // CONFIGURAÇÃO DA SÍNTESE DE VOZ (J.A.R.V.I.S.)
    const synth = window.speechSynthesis;
    
    function falarJarvis(texto) {
      if (synth.speaking) synth.cancel();

      const statusText = document.getElementById('statusText');
      const reactor = document.getElementById('arcReactor');

      statusText.innerText = `J.A.R.V.I.S.: "${texto}"`;
      reactor.classList.add('speaking');

      const utterance = new SpeechSynthesisUtterance(texto);
      utterance.lang = 'pt-BR';
      utterance.rate = 0.95; // Tom cadenciado e calmo
      utterance.pitch = 0.8; // Tom mais grave para lembrar o J.A.R.V.I.S.

      utterance.onend = () => {
        reactor.classList.remove('speaking');
        statusText.innerText = "Aguardando seu comando...";
      };

      synth.speak(utterance);
    }

    // CONFIGURAÇÃO DO RECONHECIMENTO DE VOZ (ALEXA STYLE)
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    let recognition = null;

    if (SpeechRecognition) {
      recognition = new SpeechRecognition();
      recognition.lang = 'pt-BR';
      recognition.continuous = false;

      recognition.onstart = () => {
        reconhecendo = true;
        document.getElementById('arcReactor').classList.add('listening');
        document.getElementById('statusText').innerText = "Escutando comandos...";
      };

      recognition.onresult = (event) => {
        const comando = event.results[0][0].transcript.toLowerCase();
        processarComando(comando);
      };

      recognition.onend = () => {
        reconhecendo = false;
        document.getElementById('arcReactor').classList.remove('listening');
      };
    } else {
      alert("Navegador não suporta reconhecimento de voz direto. Recomenda-se o uso do Google Chrome.");
    }

    function alternarEscuta() {
      if (!recognition) return;
      if (reconhecendo) {
        recognition.stop();
      } else {
        recognition.start();
      }
    }

    // PROCESSAMENTO DE COMANDOS DE VOZ
    function processarComando(comando) {
      document.getElementById('statusText').innerText = `Você falou: "${comando}"`;

      if (comando.includes("música") || comando.includes("spotify")) {
        falarJarvis("Iniciando playlist motivacional no Spotify agora mesmo, Senhor.");
        abrirSpotify("workout playlist");
      } 
      else if (comando.includes("treino em casa") || comando.includes("casa")) {
        document.getElementById('ambienteTreino').value = "Casa";
        falarJarvis("Modo de treino em casa ativado. Sugiro exercícios de calistenia, polichinelos e flexões.");
      } 
      else if (comando.includes("academia")) {
        document.getElementById('ambienteTreino').value = "Academia";
        falarJarvis("Modo academia selecionado. Não se esqueça de registrar as cargas e respeitar os tempos de descanso.");
      } 
      else if (comando.includes("meta")) {
        falarJarvis(metaAtual ? `Sua meta atual é: ${metaAtual}` : "Você ainda não definiu nenhuma meta, Senhor.");
      }
      else if (comando.includes("horas") || comando.includes("horário")) {
        const agora = new Date();
        falarJarvis(`Agora são exatamente ${agora.getHours()} horas e ${agora.getMinutes()} minutos.`);
      }
      else {
        falarJarvis("Comando recebido, Senhor. Estou ajustando os parâmetros do seu sistema de treino.");
      }
    }

    // MÉTODOS DE MANIPULAÇÃO DA INTERFACE
    function definirMeta() {
      const input = document.getElementById('metaInput');
      if (input.value.trim() !== "") {
        metaAtual = input.value;
        document.getElementById('metaAtualText').innerText = `Meta: ${metaAtual}`;
        falarJarvis(`Meta registrada com sucesso: ${metaAtual}`);
        input.value = "";
      }
    }

    function adicionarAlarme() {
      const desc = document.getElementById('descricaoAlarme').value;
      const hora = document.getElementById('horaAlarme').value;

      if (!desc || !hora) {
        alert("Preencha a descrição e o horário!");
        return;
      }

      const alarme = { id: Date.now(), desc, hora };
      alarmes.push(alarme);
      atualizarListaAlarmes();
      falarJarvis(`Alarme configurado para as ${hora}.`);

      document.getElementById('descricaoAlarme').value = "";
      document.getElementById('horaAlarme').value = "";
    }

    function atualizarListaAlarmes() {
      const ul = document.getElementById('listaAlarmes');
      ul.innerHTML = "";
      alarmes.forEach(a => {
        ul.innerHTML += `
          <li>
            <span>⏰ <b>${a.hora}</b> - ${a.desc}</span>
            <button class="delete-btn" onclick="removerAlarme(${a.id})">Remover</button>
          </li>
        `;
      });
    }

    function removerAlarme(id) {
      alarmes = alarmes.filter(a => a.id !== id);
      atualizarListaAlarmes();
    }

    function abrirSpotify(termo) {
      const url = `https://open.spotify.com/search/${encodeURIComponent(termo)}`;
      window.open(url, '_blank');
    }

    // SISTEMA CONTINUO DE CHECAGEM DE ALARMES / DESPERTADOR
    setInterval(() => {
      const agora = new Date();
      const horaAtual = `${String(agora.getHours()).padStart(2, '0')}:${String(agora.getMinutes()).padStart(2, '0')}`;
      
      alarmes.forEach((a, index) => {
        if (a.hora === horaAtual && !a.tocado) {
          a.tocado = true;
          falarJarvis(`Atenção, Senhor! Hora do seu compromisso: ${a.desc}`);
          alert(`⏰ J.A.R.V.I.S. ALARME: ${a.desc}`);
        }
      });
    }, 10000); // Checa a cada 10 segundos
  </script>
